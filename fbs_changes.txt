STEP 1 — ADD getLifeStage(user) HELPER

function getLifeStage(user) {
  const age = user.age
  if (age < 30)  return 'EARLY_CAREER'
  if (age < 40)  return 'ESTABLISHING'
  if (age < 50)  return 'CONSOLIDATING'
  if (age < 58)  return 'PEAK_EARNING'
  return 'PRE_RETIREMENT'
}
